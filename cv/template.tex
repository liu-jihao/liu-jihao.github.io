\documentclass[11pt]{article}

% ---------- Page geometry ----------
\usepackage[
  letterpaper,
  top=0.85in,
  bottom=0.95in,
  left=0.95in,
  right=0.95in,
  footskip=0.5in
]{geometry}

% ---------- Fonts (XeLaTeX) ----------
\usepackage{amssymb}
\usepackage{fontspec}
% Load the same EB Garamond faces directly. The package also loads an unused
% decorative initials face, which produces misleading missing-space warnings.
\setmainfont{EBGaramond}[
  Extension=.otf, UprightFont=*-Regular, ItalicFont=*-Italic,
  BoldFont=*-Bold, BoldItalicFont=*-BoldItalic,
  Numbers={Proportional,Lining}, Ligatures=TeX]
% Preserve the original macOS font. TeX Live's Fandol Song is a portable fallback.
% A standalone build may set \CVChineseFont before \input{cv.tex} to override it.
\ifdefined\CVChineseFont
  \newfontfamily\cnfont{\CVChineseFont}[Scale=0.95]
\else
  \IfFontExistsTF{Songti SC}
    {\newfontfamily\cnfont{Songti SC}[Scale=0.95]}
    {\newfontfamily\cnfont{FandolSong-Regular.otf}[Scale=0.95]}
\fi

% ---------- Color (same palette as the website) ----------
\usepackage[dvipsnames,table]{xcolor}
\definecolor{accent}{HTML}{8B1A1A}      % academic red, only for headings + name
\definecolor{textcol}{HTML}{1A1A1A}
\definecolor{muted}{HTML}{555555}
\definecolor{rule}{HTML}{C9C5B8}
\definecolor{linkcol}{HTML}{6B1414}     % darker red for arXiv/DOI links

% ---------- Hyperlinks ----------
\usepackage[hidelinks]{hyperref}
\usepackage{xurl}
\usepackage{bookmark}
\hypersetup{
  pdftitle    = {Jihao Liu -- Curriculum Vitae},
  pdfauthor   = {Jihao Liu},
  pdfsubject  = {Curriculum Vitae},
  pdfkeywords = {algebraic geometry, birational geometry, foliations,
                 minimal model program, singularities},
  colorlinks  = true,
  linkcolor   = linkcol,
  urlcolor    = linkcol,
  citecolor   = linkcol,
  breaklinks  = true,
  pdflang     = {en},
}
\urlstyle{same}
% A digest of the generated source gives the same trailer ID in every build
% directory. The build sets SOURCE_DATE_EPOCH for the PDF creation timestamp.
\AtBeginDocument{\special{pdf:trailerid [<@@PDF_ID@@> <@@PDF_ID@@>]}}

% ---------- Section heading style ----------
\usepackage{titlesec}
\titleformat{\section}
  {\normalfont\large\scshape\color{accent}}{}{0pt}
  {}[\vspace{2pt}{\color{rule}\titlerule[0.4pt]}]
\titlespacing*{\section}{0pt}{1.2\baselineskip}{0.5\baselineskip}

\titleformat{\subsection}
  {\normalfont\normalsize\scshape\color{textcol}}{}{0pt}{}
\titlespacing*{\subsection}{0pt}{0.6\baselineskip}{0.2\baselineskip}

% ---------- Header / footer ----------
\usepackage{fancyhdr}
\usepackage{lastpage}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[L]{\small\color{muted}Jihao Liu \;\textbf{\textperiodcentered}\; CV}
\fancyfoot[R]{\small\color{muted}\thepage\ / \pageref{LastPage}}

% First page: no header, no footer
\fancypagestyle{firststyle}{
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \fancyfoot[R]{\small\color{muted}\thepage\ / \pageref{LastPage}}
}

% ---------- Lists ----------
\usepackage{enumitem}
\setlist{nosep,leftmargin=*}
\setlength{\emergencystretch}{1em} % prevent tiny overruns in long journal names

% ---------- Page-break helpers ----------
\usepackage{needspace}
\newcommand{\cvsection}[2]{%
  \Needspace{5\baselineskip}%
  \pdfbookmark[1]{#1}{#2}%
  \section*{#1}%
}
\newcommand{\cvsubsection}[2]{%
  \Needspace{4\baselineskip}%
  \pdfbookmark[2]{#1}{#2}%
  \subsection*{#1}%
}
% Avoid a lonely final line in longer prose outside the unbreakable entries.
\clubpenalty=10000
\widowpenalty=10000

% ---------- Two-column "date | entry" layout ----------
% \entry{date span}{title and details}
\newlength{\datewidth}
\setlength{\datewidth}{1.4in}
\newcommand{\entry}[2]{%
  \noindent
  \begin{minipage}[t]{\datewidth}
    \raggedright\small\color{muted}#1
  \end{minipage}%
  \hfill%
  \begin{minipage}[t]{\dimexpr\linewidth-\datewidth-0.1in\relax}
    \raggedright #2
  \end{minipage}\par\vspace{4pt}
}

% Numbered publication entry (uses an automatic counter so we can
% renumber if the order changes -- but the absolute number on the
% website is stable, so we set it manually with \pubitem{number}).
\newcommand{\pubitem}[3]{%
  \noindent
  \begin{minipage}[t]{\datewidth}
    \raggedright\small\color{muted}#1.\;\textit{#2}
  \end{minipage}%
  \hfill%
  \begin{minipage}[t]{\dimexpr\linewidth-\datewidth-0.1in\relax}
    \raggedright #3
  \end{minipage}\par\vspace{4pt}
}

% Inline emphasis for paper titles
\newcommand{\paper}[1]{\textbf{#1}}
% Light separator dot for inline metadata
\newcommand{\sep}{\,\textperiodcentered\,}
% Author rendering: bold the author's own name
\newcommand{\me}{\textbf{Jihao Liu}}
% Preserve readable identifiers while permitting breaks in long DOI strings.
\newcommand{\idlink}[2]{\href{#1}{\nolinkurl{#2}}}

\usepackage{titling}
